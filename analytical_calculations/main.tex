\section{Gaussian covariance derivation}
\label{sec:intro}
In general, the estimator for the power spectrum between fields \(\alpha\) and \(\beta\) can be written as (here, ignoring the shot-noise contribution) 
\begin{equation}
    P^{\alpha\beta}_\ell(\boldsymbol{k}) = \frac{(2l+1)}{I_{22}^{\alpha\beta}}\int_{\hat{\boldsymbol{k}}} \frac{1}{2}\left(F_\ell^{\alpha}(\boldsymbol{k})F_0^{\beta, *}(\boldsymbol{k}) + F_0^{\alpha}(\boldsymbol{k})F_\ell^{\beta, *}(\boldsymbol{k})\right),
    \label{eq: P_general}
\end{equation}
where
\begin{equation}
    F^{\gamma}_\ell(\boldsymbol{k}) = \int_{\boldsymbol{x}}W_{11}^{\gamma}(\boldsymbol{x}) e^{-i\boldsymbol{k}\cdot\boldsymbol{x}}\mathcal{L}_\ell(\hat{\boldsymbol{k}} \cdot \hat{\boldsymbol{x}}) \gamma(\boldsymbol{x})
    \label{eq:F}
\end{equation}
and
\begin{equation}
    I_{22}^{\alpha\beta} = \int_{\boldsymbol{x}}\bar{n}_{\alpha}(\boldsymbol{x})\bar{n}_{\beta}(\boldsymbol{x})w_{\alpha}(\boldsymbol{x})w_{\beta}(\boldsymbol{x})
\end{equation}
for an arbitrary field \(\gamma\) and it has the following property
\begin{equation}
    F_\ell^{\gamma,*}(-\boldsymbol{k}) = (-1)^\ell F_\ell^\gamma(\boldsymbol{k})
    \label{eq:F_prop}
\end{equation}
since the field \(\gamma\) and its corresponding weights \(W_{11}^\gamma\) are real. The Gaussian covariance matrix of the power spectrum is given by 
\begin{align}
G_{\ell_1 \ell_2}^{a b c d}(k_1, k_2) &= \langle P^{ab}_{\ell_1}P^{cd}_{\ell_2}\rangle - \langle P^{ab}_{\ell_1}\rangle \langle P^{cd}_{\ell_2}\rangle.
\label{eq:G_start}
\end{align}
Substituting equation~(\ref{eq: P_general}) into equation~(\ref{eq:G_start}), we get 
\begin{align}
    G_{\ell_1 \ell_2}^{a b c d}(k_1, k_2) &= \frac{(2\ell_1 + 1)(2\ell_2 + 1)}{4I_{22}^{ab} I_{22}^{cd}}\int_{\hat{\boldsymbol{k}}_1, \hat{\boldsymbol{k}}_2} \Big(\Big[\big\langle F_{\ell_1}^a(\boldsymbol{k}_1) F_0^{b, *}(\boldsymbol{k}_1) F_{\ell_2}^c(\boldsymbol{k}_2) F_0^{d, *}(\boldsymbol{k}_2)\big\rangle \nonumber \\
    &- \langle F_{\ell_1}^a(\boldsymbol{k}_1)F_0^{b, *}(\boldsymbol{k}_1) \rangle \langle F_{\ell_2}^{c}(\boldsymbol{k}_2)F_0^{d, *}(\boldsymbol{k}_2) \big\rangle\Big] + \Big[\big\langle F_{\ell_1}^a(\boldsymbol{k}_1) F_0^{b, *}(\boldsymbol{k}_1) F_{0}^c(\boldsymbol{k}_2) F_{\ell_2}^{d, *}(\boldsymbol{k}_2)\big\rangle \nonumber \\
    &- \langle F_{\ell_1}^a(\boldsymbol{k}_1)F_0^{b, *}(\boldsymbol{k}_1) \rangle \langle F_{0}^{c}(\boldsymbol{k}_2)F_{\ell_2}^{d, *}(\boldsymbol{k}_2) \big\rangle\Big] + \Big[\big\langle F_{0}^a(\boldsymbol{k}_1) F_{\ell_1}^{b, *}(\boldsymbol{k}_1) F_{\ell_2}^c(\boldsymbol{k}_2) F_0^{d, *}(\boldsymbol{k}_2)\big\rangle \nonumber \\
    &- \langle F_{0}^a(\boldsymbol{k}_1)F_{\ell_1}^{b, *}(\boldsymbol{k}_1) \rangle \langle F_{\ell_2}^{c}(\boldsymbol{k}_2)F_0^{d, *}(\boldsymbol{k}_2) \big\rangle\Big] + \Big[\big\langle F_{0}^a(\boldsymbol{k}_1) F_{\ell_1}^{b, *}(\boldsymbol{k}_1) F_{0}^c(\boldsymbol{k}_2) F_{\ell_2}^{d, *}(\boldsymbol{k}_2)\big\rangle \nonumber \\
    &- \langle F_{0}^a(\boldsymbol{k}_1)F_{\ell_1}^{b, *}(\boldsymbol{k}_1) \rangle \langle F_{0}^{c}(\boldsymbol{k}_2)F_{\ell_2}^{d, *}(\boldsymbol{k}_2) \big\rangle\Big] \Big)
    \label{eq: G_expand}
\end{align}
We will focus on deriving the first term. The expressions for the other three terms can be inferred based on the first term using symmetry arguments. Applying Wick's theorem for Gaussian random fields, for the first term, we get 
\begin{align}
\mathcal{G}_{\ell_1 \ell_2}^{a b c d}(\boldsymbol{k_1}, \boldsymbol{k_2}) &=\frac{(2\ell_1 + 1)(2\ell_2 + 1)}{4I_{22}^{ab} I_{22}^{cd}}\int_{\hat{\boldsymbol{k}}_1, \hat{\boldsymbol{k}}_2} \Big[\big\langle F_{\ell_1}^a(\boldsymbol{k}_1) \big(F_0^d(\boldsymbol{k}_2)\big)^* \big\rangle \big\langle F_{\ell_2}^c(\boldsymbol{k}_2) \big(F_0^b(\boldsymbol{k}_1)\big)^* \big\rangle \nonumber\\
&+ \big\langle F_{\ell_1}^a(\boldsymbol{k}_1) F_{\ell_2}^{c}(\boldsymbol{k}_2) \big\rangle \big\langle (F_0^b(\boldsymbol{k}_1))^{\star} \big(F_0^d(\boldsymbol{k}_2)\big)^* \big\rangle \Big] \nonumber\\
\nonumber\\
&= \frac{(2\ell_1 + 1)(2\ell_2 + 1)}{I_{22}^{ab} I_{22}^{cd}}\int_{\hat{\boldsymbol{k}}_1, \hat{\boldsymbol{k}}_2} \int_{\boldsymbol{x_1}, \boldsymbol{x_1}', \boldsymbol{x_2}, \boldsymbol{x_2}'} e^{-i \boldsymbol{k_1} \cdot (\boldsymbol{x_1} - \boldsymbol{x_1}') - i \boldsymbol{k_2} \cdot (\boldsymbol{x_2} - \boldsymbol{x_2}')} L_{\ell_1}(\hat{\boldsymbol{k}}_1 \cdot \hat{\boldsymbol{x}}_1) \nonumber\\
& \Bigg\{W_{11}^a(\boldsymbol{x_1}) W_{11}^b(\boldsymbol{x_1}') W_{11}^c(\boldsymbol{x_2}) W_{11}^d(\boldsymbol{x_2}')
\langle a(\boldsymbol{x_1}) d(\boldsymbol{x_2}') \rangle \langle c(\boldsymbol{x_2}) b(\boldsymbol{x_1}') \rangle L_{\ell_2}(\hat{\boldsymbol{k}}_2 \cdot \hat{\boldsymbol{x}}_2) \nonumber\\
&\quad + W_{11}^a(\boldsymbol{x_1}) W_{11}^b(\boldsymbol{x_1}') W_{11}^c(\boldsymbol{x_2}') W_{11}^d(\boldsymbol{x_2})\langle a(\boldsymbol{x_1}) c(\boldsymbol{x_2}') \rangle \langle b(\boldsymbol{x_1}') d(\boldsymbol{x_2}) \rangle L_{\ell_2}(-\hat{\boldsymbol{k}}_2 \cdot \hat{\boldsymbol{x}}_2') \Bigg\}
\label{eq:G_sub}
\end{align}
For the second part of equation~(\ref{eq:G_sub}), we substitute \(\boldsymbol{x_2} \leftrightarrow \boldsymbol{x_2}'\) and \(\boldsymbol{k_2} \rightarrow -\boldsymbol{k_2}\) (These are variables are being integrated over, so we can swap them freely).

Substituting \(\boldsymbol{s_1} = \boldsymbol{x_1}-\boldsymbol{x_2}'\) and \(\boldsymbol{s_2} = \boldsymbol{x_2}-\boldsymbol{x_1}'\), we have
\begin{equation}
\begin{split}
\mathcal{G}^{abcd}_{\ell_1 \ell_2}(k_1, k_2) &= \frac{(2\ell_1 + 1)(2\ell_2 + 1)}{4I_{22}^{ab} I_{22}^{cd}} \int_{\hat{\mathbf{k}}_1, \hat{\mathbf{k}}_2} \int_{\mathbf{x}_1, \mathbf{x}_2, \mathbf{s}_1, \mathbf{s}_2} e^{-i(\mathbf{k}_1 - \mathbf{k}_2) \cdot (\mathbf{x}_1 - \mathbf{x}_2)} e^{-i\mathbf{k}_1 \cdot \mathbf{s}_2 - i\mathbf{k}_2 \cdot \mathbf{s}_1}  \nonumber\\
&\quad \times\Big[ W^a_{11}(\mathbf{x}_1)W^d_{11}(\mathbf{x}_1 - \mathbf{s}_1)W^c_{11}(\mathbf{x}_2)W^b_{11}(\mathbf{x}_2 - \mathbf{s}_2)\mathcal{L}_{\ell_1}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_1) \nonumber\\
& \times\xi_{ad}(\mathbf{s}_1; \mathbf{x}_1)\xi_{bc}(\mathbf{s}_2; \mathbf{x}_2)\mathcal{L}_{\ell_2}(\hat{\mathbf{x}}_2 \cdot \hat{\mathbf{k}}_2) + W^a_{11}(\mathbf{x}_1)W^c_{11}(\mathbf{x}_1 - \mathbf{s}_1)W^d_{11}(\mathbf{x}_2)W^b_{11}(\mathbf{x}_2 - \mathbf{s}_2) \nonumber\\
& \times \xi_{ac}(\mathbf{s}_1; \mathbf{x}_1)\xi_{bd}(\mathbf{s}_2; \mathbf{x}_2)\mathcal{L}_{\ell_1}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_1)(-1)^{\ell_2}\mathcal{L}_{\ell_2}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_2)]\Big].
\end{split}
\end{equation}

In the last line, we assume the local plane parallel approximation such that \(\boldsymbol{x_1} - \boldsymbol{s_1} = \boldsymbol{x_1}\). Integrating over \(\boldsymbol{s_1}\) and \(\boldsymbol{s_2}\), we will get

\begin{align}
\mathcal{G}^{abcd}_{\ell_1 \ell_2}(k_1, k_2) 
&\simeq \sum_{\ell'_1, \ell'_2} \Bigg\{\frac{(2\ell_1 + 1)(2\ell_2 + 1)}{4I_{22}^{ab} I_{22}^{cd}} \int_{\hat{\mathbf{k}}_1, \hat{\mathbf{k}}_2, \mathbf{x}_1, \mathbf{x}_2} e^{-i(\mathbf{x}_1 - \mathbf{x}_2) \cdot (\mathbf{k}_1 - \mathbf{k}_2)} \nonumber\\
&\quad \times \mathcal{L}_{\ell_1}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_1)\mathcal{L}_{\ell'_2}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_2)\mathcal{L}_{\ell'_1}(\hat{\mathbf{x}}_2 \cdot \hat{\mathbf{k}}_1)[W_{22}^{ad}(\mathbf{x}_1) W_{22}^{bc}(\mathbf{x}_2)P^{ad}_{\ell'_2}(k_2) P^{bc}_{\ell'_1}(k_1) \mathcal{L}_{\ell_2}(\hat{\mathbf{x}}_2 \cdot \hat{\mathbf{k}}_2) \nonumber\\
&+ W_{22}^{ac}(\mathbf{x}_1) W_{22}^{bd}(\mathbf{x}_2)P^{ac}_{\ell'_2}(k_2) P^{bd}_{\ell'_1}(k_1)(-1)^{\ell_2}\mathcal{L}_{\ell_2}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_2)] \Bigg\}, \nonumber\\
&\equiv \sum_{\ell'_1, \ell'_2} P^{ad}_{\ell'_2}(k_2) P^{bc}_{\ell'_1}(k_1) \mathcal{W}^{(ad, bc)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_2, x_1) + \nonumber \\
&\sum_{\ell'_1, \ell'_2} P^{ac}_{\ell'_2}(k_2) P^{bd}_{\ell'_1}(k_1) (-1)^{\ell_2}\mathcal{W}^{(ac, bd)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_1, x_1)
\end{align}
where
\begin{align}
    \mathcal{W}^{(\alpha \beta, \gamma \epsilon)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_{\zeta}, x_\eta) &= \frac{(2\ell_1 + 1)(2\ell_2 + 1)}{4I_{22}^{ab} I_{22}^{cd}} \int_{\hat{\mathbf{k}}_1, \hat{\mathbf{k}}_2, \mathbf{x}_1, \mathbf{x}_2} e^{-i(\mathbf{x}_1 - \mathbf{x}_2) \cdot (\mathbf{k}_1 - \mathbf{k}_2)} \nonumber\\
    & \times \mathcal{L}_{\ell_1}(\hat{\mathbf{x}}_\eta \cdot \hat{\mathbf{k}}_1)\mathcal{L}_{\ell'_1}(\hat{\mathbf{x}}_2 \cdot \hat{\mathbf{k}}_1)\mathcal{L}_{\ell'_2}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_2)W_{22}^{\alpha\epsilon}(\mathbf{x}_1) W_{22}^{\beta\gamma}(\mathbf{x}_2)\mathcal{L}_{\ell_2}(\hat{\mathbf{x}}_\zeta \cdot \hat{\mathbf{k}}_2)
\end{align}

Using symmetry in equations~(\ref{eq: G_expand}), the full expression for the Gaussian covariance matrix is 
\begin{align}
    G_{\ell_1 \ell_2}^{a b c d}(k_1, k_2) &= \sum_{\ell'_1, \ell'_2} P^{ad}_{\ell'_2}(k_2) P^{bc}_{\ell'_1}(k_1) \mathcal{W}^{(ad, bc)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_2, x_1) + \nonumber \\
    &\sum_{\ell'_1, \ell'_2} P^{ac}_{\ell'_2}(k_2) P^{bd}_{\ell'_1}(k_1) (-1)^{\ell_2}\mathcal{W}^{(ac, bd)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_1, x_1) \nonumber \\
    &+\sum_{\ell'_1, \ell'_2} P^{ad}_{\ell'_2}(k_2) P^{bc}_{\ell'_1}(k_1) \mathcal{W}^{(ad, bc)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_1, x_1) + \nonumber \\
    &\sum_{\ell'_1, \ell'_2} P^{ac}_{\ell'_2}(k_2) P^{bd}_{\ell'_1}(k_1) (-1)^{\ell_2}\mathcal{W}^{(ac, bd)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_2, x_1) \nonumber \\
    &\sum_{\ell'_1, \ell'_2} P^{ad}_{\ell'_2}(k_2) P^{bc}_{\ell'_1}(k_1) \mathcal{W}^{(ad, bc)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_2, x_2) + \nonumber \\
    &\sum_{\ell'_1, \ell'_2} P^{ac}_{\ell'_2}(k_2) P^{bd}_{\ell'_1}(k_1) (-1)^{\ell_2}\mathcal{W}^{(ac, bd)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_1, x_2) \nonumber \\
    &\sum_{\ell'_1, \ell'_2} P^{ad}_{\ell'_2}(k_2) P^{bc}_{\ell'_1}(k_1) \mathcal{W}^{(ad, bc)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_1, x_2) + \nonumber \\
    &\sum_{\ell'_1, \ell'_2} P^{ac}_{\ell'_2}(k_2) P^{bd}_{\ell'_1}(k_1) (-1)^{\ell_2}\mathcal{W}^{(ac, bd)}_{\ell_1, \ell_2, \ell'_1, \ell'_2}(k_1, k_2, x_2, x_2)
    \label{eq:G_all}
\end{align}

For the auto-covariance matrix of the galaxy power spectrum multipole, \(a=b=c=d=g\) and \(\ell_1, \ell_2\) are even. Equation~(\ref{eq:G_all}) reduces to 
\begin{align}
    G_{\ell_1 \ell_2}^{a b c d}(k_1, k_2) &= \sum_{\ell'_1, \ell'_2} P^{gg}_{\ell'_2}(k_2) P^{gg}_{\ell'_1}(k_1)\frac{(2\ell_1 + 1)(2\ell_2 + 1)}{4I_{22}^{gg} I_{22}^{gg}} \int_{\hat{\mathbf{k}}_1, \hat{\mathbf{k}}_2, \mathbf{x}_1, \mathbf{x}_2} e^{-i(\mathbf{x}_1 - \mathbf{x}_2) \cdot (\mathbf{k}_1 - \mathbf{k}_2)}W_{22}^{gg}(x_1)W_{22}^{gg}(x_2) \nonumber \\
    &\mathcal{L}_{\ell'_1}(\hat{\mathbf{x}}_2 \cdot \hat{\mathbf{k}}_1)\mathcal{L}_{\ell'_2}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_2)\Big[\mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_2) + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_2) \nonumber \\
    & + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_2) + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_2)\nonumber \\ 
    &+\mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_2) + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_2) \nonumber \\
    &+ \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_2) + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_2)\Big] \nonumber \\
    &= \sum_{\ell'_1, \ell'_2} P^{gg}_{\ell'_2}(k_2) P^{gg}_{\ell'_1}(k_1)\frac{(2\ell_1 + 1)(2\ell_2 + 1)}{2I_{22}^{gg} I_{22}^{gg}} \int_{\hat{\mathbf{k}}_1, \hat{\mathbf{k}}_2, \mathbf{x}_1, \mathbf{x}_2} e^{-i(\mathbf{x}_1 - \mathbf{x}_2) \cdot (\mathbf{k}_1 - \mathbf{k}_2)}W_{22}^{gg}(x_1)W_{22}^{gg}(x_2) \nonumber \\
    &\mathcal{L}_{\ell'_1}(\hat{\mathbf{x}}_2 \cdot \hat{\mathbf{k}}_1)\mathcal{L}_{\ell'_2}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_2)\Big[\mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_2) + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_2) \nonumber \\
    &+ \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_2) + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_2)\Big] \nonumber \\
    &=\sum_{\ell'_1, \ell'_2} P^{gg}_{\ell'_2}(k_2) P^{gg}_{\ell'_1}(k_1)\frac{(2\ell_1 + 1)(2\ell_2 + 1)}{I_{22}^{gg} I_{22}^{gg}} \int_{\hat{\mathbf{k}}_1, \hat{\mathbf{k}}_2, \mathbf{x}_1, \mathbf{x}_2} e^{-i(\mathbf{x}_1 - \mathbf{x}_2) \cdot (\mathbf{k}_1 - \mathbf{k}_2)}W_{22}^{gg}(x_1)W_{22}^{gg}(x_2) \nonumber \\
    &\mathcal{L}_{\ell'_1}(\hat{\mathbf{x}}_2 \cdot \hat{\mathbf{k}}_1)\mathcal{L}_{\ell'_2}(\hat{\mathbf{x}}_1 \cdot \hat{\mathbf{k}}_2)\Big[\mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_2 \cdot \hat{\boldsymbol{k}}_2) + \mathcal{L}_{\ell_1}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_1)\mathcal{L}_{\ell_2}(\hat{\boldsymbol{x}}_1 \cdot \hat{\boldsymbol{k}}_2)\Big]
\end{align}
consistent with ref.~\citep{Wadekar_2020}. In the third equality, we swap \(x_1 \leftrightarrow x_2\), \(k_1 \rightarrow -k_1\) and \(k_2 \rightarrow -k_2\) for the third and fourth terms since they are variables being integrated over. 

\section{Shot-noise for the Gaussian covariance matrix}
Following Appendix B of ref.~\citep{Wadekar_2020}, the discrete analogue of the momentum field estimator should be 
\begin{equation}
    F_\ell^p(\boldsymbol{k}) = \sum_i^{N_p} w_i^p(\boldsymbol{x})u_i(\boldsymbol{x})e^{-i\boldsymbol{k}\cdot\boldsymbol{x}} L(\hat{\boldsymbol{k}} \cdot \hat{\boldsymbol{x}})
\end{equation}
using the definition from ref.~\citep{Howlett_2019}. Notice that \(N_p\) denotes the number of galaxies with peculiar velocity measurements and \(w^p\) is the FKP weight for the momentum field \citep{Howlett_2019}
\begin{equation}
    w^p(\boldsymbol{x}) = \frac{1}{u^2(\boldsymbol{x}) + \bar{n}_p(\boldsymbol{x})P^p(\boldsymbol{k}, \boldsymbol{x})}.
\end{equation}

% \subsection{Momentum}

The auto-power spectrum of the momentum can be represented as
\begin{align}
    \langle F^p_{\ell_1}(\boldsymbol{k}_1) F^p_{\ell_2}(\boldsymbol{k}_2)\rangle
    &= \langle \sum_i^{N_g}\sum_j^{N_g} w_i^p w_j^p u_i u_j e^{-i\boldsymbol{k}_1\cdot\boldsymbol{x}_i-i\boldsymbol{k}_2\cdot\boldsymbol{x}_j}
    \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}}_1\cdot\hat{\boldsymbol{x}}_i) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}}_2\cdot\hat{\boldsymbol{x}}_j) \rangle \nonumber \\
    &= \langle\sum_i^{N_g}\sum_{j\neq i}^{N_g} w_i^p w_j^p u_i u_j e^{-i\boldsymbol{k}_1\cdot\boldsymbol{x}_i-i\boldsymbol{k}_2\cdot\boldsymbol{x}_j}
    \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}}_1\cdot\hat{\boldsymbol{x}}_i) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}}_2\cdot\hat{\boldsymbol{x}}_j) \rangle \nonumber \\
    &+\langle\sum_i^{N_g} {w_i^p}^2 u_i^2 e^{-i(\boldsymbol{k}_1+\boldsymbol{k}_2)\cdot\boldsymbol{x}_i}
    \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}}_1\cdot\hat{\boldsymbol{x}}_i) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}}_2\cdot\hat{\boldsymbol{x}}_i) \rangle \nonumber \\
    &\simeq \int_{\boldsymbol{x}_1, \boldsymbol{x}_2} W^p_{11}(\boldsymbol{x}_1) W^p_{11}(\boldsymbol{x}_2) \langle (1+\delta(\boldsymbol{x}_1))u(\boldsymbol{x}_1) (1+\delta(\boldsymbol{x}_2))u(\boldsymbol{x}_2)\rangle
    e^{-i\boldsymbol{k}_1\cdot\boldsymbol{x}_1-i\boldsymbol{k}_2\cdot\boldsymbol{x}_2} \\
    & \times \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}}_1\cdot\hat{\boldsymbol{x}}_1) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}}_2\cdot\hat{\boldsymbol{x}}_2) \nonumber \\
    &+ \int_{\boldsymbol{x}} W^p_{12}(\boldsymbol{x})\langle u^2(\boldsymbol{x})\rangle e^{-i(\boldsymbol{k}_1+\boldsymbol{k}_2)\cdot\boldsymbol{x}}
    \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}}_1\cdot\hat{\boldsymbol{x}})
    \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}}_2\cdot\hat{\boldsymbol{x}}) \nonumber \\
    & = \langle F^p_{\ell_1}(\boldsymbol{k}_1) F^p_{\ell_2}(\boldsymbol{k}_2) \rangle_{\mathrm{cont}} + {\mathcal{I}_{12}^p}^{\ell_1, \ell_2} (\boldsymbol{k}_1, \boldsymbol{k}_2)
\end{align}

The continuous integral is
\begin{align}
    \langle F^p_{\ell_1}(\boldsymbol{k}_1) F^p_{\ell_2}(\boldsymbol{k}_2) \rangle_{\mathrm{cont}}
    & =
    \int_{\boldsymbol{x}_1, \boldsymbol{x}_2} W^p_{11}(\boldsymbol{x}_1) W^p_{11}(\boldsymbol{x}_2) \langle (1+\delta(\boldsymbol{x}_1))u(\boldsymbol{x}_1) (1+\delta(\boldsymbol{x}_2))u(\boldsymbol{x}_2)\rangle
    e^{-i\boldsymbol{k}_1\cdot\boldsymbol{x}_1-i\boldsymbol{k}_2\cdot\boldsymbol{x}_2} \\
    & \times \mathcal{L}_{\ell_1}(\boldsymbol{k}_1\cdot\boldsymbol{x}_1) \mathcal{L}_{\ell_2}(\boldsymbol{k}_2\cdot\boldsymbol{x}_2) \nonumber 
\end{align}

and the shot noise contribution of the momentum auto-power spectrum is defined as
\begin{align}
    {\mathcal{I}_{12}^p}^{\ell_1, \ell_2} (\boldsymbol{k}_1, \boldsymbol{k}_2)
    \equiv\int_{\boldsymbol{x}} W^p_{12}(\boldsymbol{x})\langle u^2(\boldsymbol{x})\rangle e^{-i(\boldsymbol{k}_1+\boldsymbol{k}_2)\cdot\boldsymbol{x}}
    \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}}_1\cdot\hat{\boldsymbol{x}}) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}}_2\cdot\hat{\boldsymbol{x}})
\end{align}

We further calculate the cross-power spectrum between galaxy density and momentum first
\begin{align}
    \langle F_{\ell_1}^d (\boldsymbol{k_1}) F_{\ell_2}^p (\boldsymbol{k_2})\rangle &= \langle (\sum_i^{N_g} - \alpha\sum_i^{N_r})\sum_j^{N_p} w_i^g w_j^p u(\boldsymbol{x}_j) e^{-i \boldsymbol{k}_1\cdot\boldsymbol{x}_i - i\boldsymbol{k}_2\cdot\boldsymbol{x}_j} \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}_1}\cdot\hat{\boldsymbol{x}_i}) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}_2}\cdot\hat{\boldsymbol{x}_j})\rangle \nonumber \\
    &=\langle (\sum_i^{N_g} - \alpha\sum_i^{N_r})\sum_{j \neq i}^{N_p} w_i^g w_j^p u(\boldsymbol{x}_j) e^{-i \boldsymbol{k}_1\cdot\boldsymbol{x}_i - i\boldsymbol{k}_2\cdot\boldsymbol{x}_j} \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}_1}\cdot\hat{\boldsymbol{x}_i}) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}_2}\cdot\hat{\boldsymbol{x}_j})\rangle \nonumber \\
    &+\langle \sum_i^{N_p}w_i^g w_i^p u(\boldsymbol{x}_i) e^{-i (\boldsymbol{k}_1+\boldsymbol{k}_2)\cdot\boldsymbol{x}_i} \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}_1}\cdot\hat{\boldsymbol{x}_i}) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}_2}\cdot\hat{\boldsymbol{x}_i})\rangle.
\end{align}
For the shot-noise contribution in the second equality, since the random catalogue does not overlap with either the galaxy catalogue or the peculiar velocity catalogue, the cross term between the peculiar velocity catalogue and the random catalogue is zero. Secondly, the only overlap between the galaxy catalogue and the peculiar velocity catalogue is galaxies that have peculiar velocity measurements. Therefore, we sum up to \(N_p\) instead of \(N_g\).\footnote{This is what has been done in the DESI PV survey, peculiar velocity measurements outside the BGS footprint are not included in the final analysis. Therefore, there will be no peculiar velocity measurements outside the BGS galaxy sample.} Focusing on the shot-noise term, in the continuous limit, it should be 
\begin{equation}
    \int_{\boldsymbol{x}} \bar{n}_p(\boldsymbol{x}) w^g(\boldsymbol{x})w^p(\boldsymbol{x}) \langle u(\boldsymbol{x}) \rangle e^{-i (\boldsymbol{k}_1+\boldsymbol{k}_2)\cdot\boldsymbol{x}} \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}_1}\cdot\hat{\boldsymbol{x}}) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}_2}\cdot\hat{\boldsymbol{x}}).
\end{equation}
Based on the cosmological principle, \(\langle u(\boldsymbol{x}) \rangle = 0\), so the cross-power spectrum has no shot-noise contribution, consistent with ref.~\citep{Howlett_2019}. Consequently, the galaxy-momentum cross-power spectrum in the continuous limit is 
\begin{equation}
    \langle F_{\ell_1}^d (\boldsymbol{k_1}) F_{\ell_2}^p (\boldsymbol{k_2)\rangle} = \int_{\boldsymbol{x}_1, \boldsymbol{x_2}} W_{11}^g(\boldsymbol{x}_1) W_{11}^p (\boldsymbol{x}_2) \langle \delta(\boldsymbol{x}_1) (1+\delta(\boldsymbol{x}_2))u(\boldsymbol{x}_2)\rangle e^{-i \boldsymbol{k}_1\cdot\boldsymbol{x}_1 - i\boldsymbol{k}_2\cdot\boldsymbol{x}_2} \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}_1}\cdot\hat{\boldsymbol{x}_1}) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}_2}\cdot\hat{\boldsymbol{x}_2})\rangle.
\end{equation}
From symmetry, the momentum galaxy cross-power spectrum is given by 
\begin{equation}
    \langle F_{\ell_1}^p (\boldsymbol{k_1}) F_{\ell_2}^d (\boldsymbol{k_2)\rangle} = \int_{\boldsymbol{x}_1, \boldsymbol{x_2}} W_{11}^g(\boldsymbol{x}_2) W_{11}^p (\boldsymbol{x}_1) \langle (1+\delta(\boldsymbol{x}_1))u(\boldsymbol{x}_1)\delta(\boldsymbol{x}_2)\rangle e^{-i \boldsymbol{k}_1\cdot\boldsymbol{x}_1 - i\boldsymbol{k}_2\cdot\boldsymbol{x}_2} \mathcal{L}_{\ell_1}(\hat{\boldsymbol{k}_1}\cdot\hat{\boldsymbol{x}_1}) \mathcal{L}_{\ell_2}(\hat{\boldsymbol{k}_2}\cdot\hat{\boldsymbol{x}_2})\rangle.
\end{equation}
